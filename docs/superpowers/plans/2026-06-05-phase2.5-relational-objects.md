# Phase 2.5 Relational Objects (Brace + Angle) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add two self-contained relational object types — `brace` (`BraceBetweenPoints`) and `angle` (`Angle`/`RightAngle`) — defined by their own object-relative points, with optional LaTeX labels, byte-identical codegen in both generators, round-trip parsing, composite Konva preview with draggable point handles, inspector panels, and sidebar buttons.

**Architecture:** Points are object-relative px (centered on `obj.x`/`obj.y`), converted to Manim units with the `polygon_free` scale (`FRAME_WIDTH`/`FRAME_HEIGHT`) and positioned by the generic post-switch `move_to`. Brace = 2 points; Angle = 3 points (vertex + 2 ends) emitted with two helper `Line`s. Optional labels emit `<var>.get_tex("...")` (non-raw, doubled-backslash escaping — same as the `latex` case, to avoid the v3.6.0 literal-`int` bug) wrapped in `VGroup`. Each emitted statement is single-line so the regex parser round-trips it.

**Tech Stack:** Vue 3 `<script setup>`, Pinia, Konva (vue-konva), Manim CE codegen (codegen.js + manim.js byte-identical), Vitest.

**Spec:** `docs/superpowers/specs/2026-06-05-phase2.5-relational-objects-design.md`.

**Parity rule (all codegen tasks):** `services/api/src/compiler/codegen.js` (`objectCode`) and `services/web/src/export/manim.js` (`objCode`) share no import; the new helper(s) and `case` arms must be **byte-identical** between them, guarded by `manim-export.test.js`. In-scope locals inside the switch (confirm by reading `case 'polygon_free'`): `n` (var name), `fill` (quoted color string), `stroke`, `sw2`, `hasFill`, `hasStroke`, `lines`, `sw`, `sh`. The generic post-switch `${n}.move_to([...])` (codegen.js ~456) positions the object — do NOT add positioning in the case.

```js
      const obj = this.project.objects.find(o => o.id === id);
      if (!obj || !['p1', 'p2', 'vertex', 'point1', 'point2'].includes(key)) return;
      if (obj[key] === undefined || !Array.isArray(pt) || pt.length !== 2) return;
```

```js
/** Escape a label for a NON-raw Python string passed to MathTex/get_tex (one backslash
    survives so LaTeX commands typeset; mirrors the `latex` case — avoids the \\int line-break bug). */
function safeLatex(s) {
  return String(s == null ? '' : s).replace(/\\/g, '\\\\').replace(/"/g, '\\"').replace(/\r?\n/g, ' ');
}
```

```js

    if (m) {
```

```js
      const baseId = varMap[base];
```

```js
function relationalHitCfg(obj) {
  const w = (obj.width || 140) * vs.value, h = (obj.height || 140) * vs.value;
  return { x: -w / 2, y: -h / 2, width: w, height: h, fill: 'rgba(76,238,249,0.04)', stroke: themeAccent.value, strokeWidth: 1, dash: [6, 4], cornerRadius: 4, listening: true };
}

function relationalLabelCfg(obj, anchor) {
  return { x: anchor[0] - 12, y: anchor[1] - 8, width: 24, text: obj.label || '', align: 'center',
    fontSize: Math.max(11, 16 * vs.value), fill: obj.fill || '#ffffff', fontStyle: 'italic', listening: false };
}

// Brace: base line p1→p2 plus a perpendicular bulge polyline (approximate).
function braceLineCfg(obj) {
  const p1 = obj.p1 || [-80, 0], p2 = obj.p2 || [80, 0];
  const z = vs.value;
  const ax = p1[0] * z, ay = p1[1] * z, bx = p2[0] * z, by = p2[1] * z;
  const dx = bx - ax, dy = by - ay, len = Math.hypot(dx, dy) || 1;
  const nx = -dy / len, ny = dx / len;            // unit normal
  const d = 14 * z;                                // bulge depth
  const mx = (ax + bx) / 2 + nx * d, my = (ay + by) / 2 + ny * d;
  return { points: [ax, ay, ax + nx * d, ay + ny * d, mx, my, bx + nx * d, by + ny * d, bx, by],
    stroke: obj.stroke || '#ffffff', strokeWidth: 2, lineJoin: 'round', tension: 0.4, listening: false };
}
function braceLabelAnchor(obj) {
  const p1 = obj.p1 || [-80, 0], p2 = obj.p2 || [80, 0];
  const z = vs.value;
  const ax = p1[0] * z, ay = p1[1] * z, bx = p2[0] * z, by = p2[1] * z;
  const dx = bx - ax, dy = by - ay, len = Math.hypot(dx, dy) || 1;
  const nx = -dy / len, ny = dx / len;
  return [(ax + bx) / 2 + nx * 26 * z, (ay + by) / 2 + ny * 26 * z];
}

// Angle: two rays from vertex.
function angleRayCfgs(obj) {
  const z = vs.value;
  const v = obj.vertex || [-40, 40], p1 = obj.point1 || [80, 40], p2 = obj.point2 || [-40, -60];
  const col = obj.stroke || '#fbbf24';
  return [
    { points: [v[0] * z, v[1] * z, p1[0] * z, p1[1] * z], stroke: col, strokeWidth: 2, listening: false },
    { points: [v[0] * z, v[1] * z, p2[0] * z, p2[1] * z], stroke: col, strokeWidth: 2, listening: false },
  ];
}
// Arc between the two rays at a pixel radius proportional to obj.radius.
function angleArcCfg(obj) {
  const z = vs.value;
  const v = obj.vertex || [-40, 40], p1 = obj.point1 || [80, 40], p2 = obj.point2 || [-40, -60];
  const a1 = Math.atan2(p1[1] - v[1], p1[0] - v[0]);
  const a2 = Math.atan2(p2[1] - v[1], p2[0] - v[0]);
  const r = (obj.radius || 0.6) / 14.222 * (stg.value.width) * z * 0.5;  // px radius ≈ manim radius scaled
  const pts = [];
  let start = a1, end = a2;
  if (end < start) end += Math.PI * 2;
  const steps = 24;
  for (let i = 0; i <= steps; i++) {
    const a = start + (end - start) * (i / steps);
    pts.push(v[0] * z + Math.cos(a) * r, v[1] * z + Math.sin(a) * r);
  }
  return { points: pts, stroke: obj.stroke || '#fbbf24', strokeWidth: 2, listening: false };
}
// Right-angle square mark at the vertex.
function angleSquareCfg(obj) {
  const z = vs.value;
  const v = obj.vertex || [-40, 40], p1 = obj.point1 || [80, 40], p2 = obj.point2 || [-40, -60];
  const u1a = Math.atan2(p1[1] - v[1], p1[0] - v[0]);
  const u2a = Math.atan2(p2[1] - v[1], p2[0] - v[0]);
  const r = 16 * z;
  const c1 = [v[0] * z + Math.cos(u1a) * r, v[1] * z + Math.sin(u1a) * r];
  const c2 = [v[0] * z + Math.cos(u2a) * r, v[1] * z + Math.sin(u2a) * r];
  const corner = [c1[0] + (c2[0] - v[0] * z), c1[1] + (c2[1] - v[1] * z)];
  return { points: [c1[0], c1[1], corner[0], corner[1], c2[0], c2[1]], stroke: obj.stroke || '#fbbf24', strokeWidth: 2, listening: false };
}
function angleLabelAnchor(obj) {
  const z = vs.value;
  const v = obj.vertex || [-40, 40], p1 = obj.point1 || [80, 40], p2 = obj.point2 || [-40, -60];
  const a1 = Math.atan2(p1[1] - v[1], p1[0] - v[0]);
  const a2 = Math.atan2(p2[1] - v[1], p2[0] - v[0]);
  let mid = (a1 + a2) / 2;
  const r = 34 * z;
  return [v[0] * z + Math.cos(mid) * r, v[1] * z + Math.sin(mid) * r];
}
```
